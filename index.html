Konfigurasi Google Spreadsheet

Masukkan URL Web App dari Apps Script yang telah Anda deploy di Spreadsheet Anda.

Cara Menghubungkan ke Spreadsheet Anda:

  1. Buat Google Spreadsheet baru di Google Drive Anda.
  2. Buka menu Ekstensi > Apps Script.
  3. Hapus seluruh kode bawaan, lalu salin (copy) kode di bawah ini:
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  
  // Ambil data JSON dari payload request
  var data = JSON.parse(e.postData.contents);
  
  // Jika kolom masih kosong, buat baris Header
  if (sheet.getLastRow() === 0) {
    var headers = [
      "Waktu Pengisian", "Wali Kelas", "Kelas", "No Absen", "Nama Siswa", 
      "Tafsir 1 (30)", "Tafsir 3 (13)", "Hadits 1 (25)", "Hadits 3 (28)", 
      "Mahfudzat 1 (86)", "Mahfudzat 3 (16)", "Doa-doa (84)", 
      "Al-Qur'an", "Wudlu", "Tayammum", "Sholat"
    ];
    sheet.appendRow(headers);
  }
  
  var timestamp = new Date();
  var rowsAdded = 0;
  
  // Loop data siswa yang dikirimkan
  for (var i = 0; i < data.siswa_list.length; i++) {
    var s = data.siswa_list[i];
    
    // Simpan baris data
    sheet.appendRow([
      timestamp,
      data.wali_kelas,
      data.kelas,
      s.no_absen,
      s.nama_siswa,
      s.tafsir_1,
      s.tafsir_3,
      s.hadits_1,
      s.hadits_3,
      s.mahfudzat_1,
      s.mahfudzat_3,
      s.doa_doa,
      s.al_quran,
      s.wudlu,
      s.tayammum,
      s.sholat
    ]);
    rowsAdded++;
  }
  
  return ContentService.createTextOutput(JSON.stringify({
    "status": "success",
    "rows_added": rowsAdded
  })).setMimeType(ContentService.MimeType.JSON);
}
  1. Klik ikon Simpan (Save).
  2. Klik tombol Terapkan (Deploy) > Penerapan Baru (New Deployment).
  3. Pilih tipe: Aplikasi Web (Web App).
  4. Setel Yang memiliki akses (Who has access) ke "Siapa saja" (Anyone).
  5. Klik Terapkan, selesaikan otorisasi akun Google Anda, lalu salin URL Aplikasi Web Anda ke form di atas.
